#Info
qemu-img info source.qcow2

------------------------------------------------------
qemu-img create [-f fmt] filename [size]
------------------------------------------------------
#Create RAW: 
qemu-img create -f raw test.img 1G

#Create QCOW2: 
qemu-img create -f qcow2 test.img 10G

------------------------------------------------------
qemu-img convert [-f fmt] [-O out_fmt] infile outfile
------------------------------------------------------
#QCOW2 to RAW:
qemu-img convert -f qcow2 -O raw /path/source.qcow2 /path/output.raw

#RAW to QCOW2
qemu-img convert -f raw -O qcow2 /path/image.img image.qcow2

------------------------------------------------------
qemu-img resize [-f fmt] filename [+/-size]
------------------------------------------------------
#Resize QCOW2 img (add 10G)
qemu-img resize disk.qcow2 +10G

#Shrink QCOW2 img
-------------------------
#1. In VM guest drive, the folowing writes zeros to unused space
dd if=/dev/zero of=zerofile bs=1M
rm zerofile

#2. Shut down VM and run on the 'zeroed' host img:
qemu-img convert -f raw -O qcow2 guest.img guest-copy.qcow2
-------------------------

#Mount/Umount
-------------------------
#Mount QCOW2 img
modprobe nbd max_part=8
qemu-nbd --connect=/dev/nbd1 /media/yourfolder/disk.qcow2
mkdir /media/diskname
mount /dev/nbd1p1 /media/diskname

#Umount QCOW2 img
umount /media/diskname
rmdir /media/diskname
qemu-nbd --disconnect /dev/nbd1
rmmod nbd
-------------------------

#add usb devices to udev rules at /etc/udev/rules.d/99-usb-stick.rules

#trigger udev
sudo udevadm control --reload-rules
sudo udevadm trigger

#--OR-- add write permission to usb devices (not persistent)
